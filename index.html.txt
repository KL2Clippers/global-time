function isWork(d,tz){
  const p=d.toLocaleTimeString('en-GB',{timeZone:tz,hour:'2-digit',minute:'2-digit',hour12:false}).split(':');
  const t=+p[0]*60+ +p[1]; return t>=540&&t<1080;
}
function getOff(tz){
  const n=new Date(),a=new Date(n.toLocaleString('en-US',{timeZone:tz})),b=new Date(n.toLocaleString('en-US',{timeZone:'Asia/Seoul'}));
  const d=Math.round((a-b)/3600000); return d>=0?`+${d}h`:`${d}h`;
}

function qs(name){document.getElementById('searchIn').value=name; doSearch();}
function doSearch(){
  const raw=document.getElementById('searchIn').value.trim();
  if(!raw) return;
  const q=raw.toLowerCase();
  let found=CITY_DB[q];
  if(!found){
    const k=Object.keys(CITY_DB).find(k=>k.includes(q)||q.includes(k));
    if(k) found=CITY_DB[k];
  }
  const res=document.getElementById('sResults');
  if(!found){
    res.innerHTML=`<div class="s-empty">😕 "<strong style="color:var(--text)">${raw}</strong>" 를 찾을 수 없습니다<br><span style="font-size:10px;color:var(--text-dim)">Try: Tokyo, London, Paris…</span></div>`;
    return;
  }
  if(!searchCities.find(c=>c.tz===found.tz&&c.name===found.name)){
    searchCities.unshift(found);
    if(searchCities.length>8) searchCities.pop();
  }
  document.getElementById('searchIn').value='';
  renderSearch();
}

    card.innerHTML=`
      <div><div class="s-city-name">${city.name}</div><div class="s-country">${city.country}</div></div>
      <div style="text-align:right;flex:1">
        <div class="s-time" id="st-${i}">--:--</div>
        <div class="s-date" id="sd-${i}">---</div>
        <div class="s-badge off" id="sb-${i}">● OFF</div>
      </div>
      <button class="s-rm" onclick="rmSearch(${i})">×</button>`;
    res.appendChild(card);
  });
  updateSearchDisplay();
}

function updateSearchDisplay(){
  const now=getNow();
  searchCities.forEach((c,i)=>{
    const tEl=document.getElementById(`st-${i}`);
    const dEl=document.getElementById(`sd-${i}`);
    const bEl=document.getElementById(`sb-${i}`);
    if(!tEl) return;
    tEl.textContent=fmtTS(now,c.tz);
    dEl.textContent=fmtD(now,c.tz);
    const w=isWork(now,c.tz);
    bEl.textContent=w?'● OPEN':'● OFF'; bEl.className=`s-badge ${w?'open':'off'}`;
  });
}
function rmSearch(i){searchCities.splice(i,1); renderSearch();}

// ─── CALCULATOR ──────────────────────────────────────────────────────────────
function applyCalc(){
  const v=document.getElementById('calcIn').value; if(!v) return;
  const [hh,mm]=v.split(':').map(Number);
  const n=new Date(), sn=new Date(n.toLocaleString('en-US',{timeZone:'Asia/Seoul'}));
  const tgt=new Date(sn); tgt.setHours(hh,mm,0,0);
  calcOffset=tgt-sn;
  document.getElementById('calcBadge').classList.add('on');
  updateAll();
}
function resetCalc(){
  calcOffset=0;
  document.getElementById('calcIn').value='';
  document.getElementById('calcBadge').classList.remove('on');
  updateAll();
}
document.getElementById('calcIn').addEventListener('keydown',e=>{if(e.key==='Enter')applyCalc();});
document.getElementById('searchIn').addEventListener('keydown',e=>{if(e.key==='Enter')doSearch();});
